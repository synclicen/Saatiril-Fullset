   125→// ─── Ratio parser ───────────────────────────────────────────────────────────
   126→function parseRatio(ratioStr: string): number {
   127→  const parts = ratioStr.split(':')
   128→  if (parts.length === 2) {
   129→    const w = parseFloat(parts[0])
   130→    const h = parseFloat(parts[1])
   131→    if (w > 0 && h > 0) return w / h
   132→  }
   133→  return 4 / 3
   134→}

   136→// ─── Helpers ────────────────────────────────────────────────────────────────
   137→// DEFENSIVE: nama/nim can be undefined if a photoHistory entry got corrupted
   138→// during reset+retake cycles. Never let .trim() crash the whole app.
   139→function sanitizeNama(nama: string | undefined | null): string {
   140→  return (nama ?? '').trim().replace(/\s+/g, '_').replace(/[^a-zA-Z0-9_]/g, '')
   141→}
   142→
   143→function sanitizeNim(nim: string | undefined | null): string {
   144→  return (nim ?? '').toString().trim().replace(/[^a-zA-Z0-9_-]/g, '')
   145→}

   147→/**
   148→ * Build a versioned filename for standard mode (Toga + Ijazah).
   149→ * version: 1 = first capture, 2+ = retake after MC reset.
   150→ * v1 → `NIM_Nama_1_Toga.jpg` (no version suffix for backwards compat)
   151→ * v2 → `NIM_Nama_1_Toga_v2.jpg`, v3 → `..._v3.jpg`, …
   152→ */
   153→function buildFilename(nim: string | undefined, nama: string | undefined, suffix: number, type: string, version: number = 1): string {
   154→  const base = `${sanitizeNim(nim)}_${sanitizeNama(nama)}_${suffix}_${type}`
   155→  return version > 1 ? `${base}_v${version}.jpg` : `${base}.jpg`
   156→}
   157→
   158→/**
   159→ * Build a versioned filename for photoshoot mode.
   160→ * version: 1 = first capture, 2+ = retake after MC reset.
   161→ * v1 → `NIM_Nama.jpg` or `NIM_Nama_Ch2.jpg` (no version suffix)
   162→ * v2 → `NIM_Nama_v2.jpg` or `NIM_Nama_Ch2_v2.jpg`, …
   163→ */
   164→function buildPhotoshootFilename(nim: string | undefined, nama: string | undefined, channel: number, version: number = 1): string {
   165→  const base = `${sanitizeNim(nim)}_${sanitizeNama(nama)}`
   166→  const withCh = channel > 1 ? `${base}_Ch${channel}` : base
   167→  return version > 1 ? `${withCh}_v${version}.jpg` : `${withCh}.jpg`
   168→}

   197→function fitAspectRatio(
   198→  availW: number,
   199→  availH: number,
   200→  ratio: number,
   201→): { width: number; height: number } {
   202→  const hFromW = availW / ratio
   203→  if (hFromW <= availH) {
   204→    return { width: availW, height: hFromW }
   205→  }
   206→  const wFromH = availH * ratio
   207→  return { width: wFromH, height: availH }
   208→}
