    27→// ─── Helpers ────────────────────────────────────────────────────────────────
    28→function isActiveStatus(status: StudentStatus): boolean {
    29→  return status.startsWith('active')
    30→}
    31→
    32→function getActiveChannel(status: StudentStatus): number | null {
    33→  if (!isActiveStatus(status)) return null
    34→  const ch = status.split('_')[1]
    35→  return ch ? parseInt(ch, 10) : null
    36→}
    37→
    38→function statusLabel(status: StudentStatus): string {
    39→  if (status === 'pending') return 'Menunggu'
    40→  if (status === 'sent') return 'Dikirim'
    41→  if (status === 'done') return 'Selesai'
    42→  const ch = getActiveChannel(status)
    43→  return ch != null ? `Foto Ch.${ch}` : 'Aktif'
    44→}

   117→    return channelStudents.find((s) => s.status === 'pending') ?? null

   120→  const remainingCount = useMemo<number>(() => {

   124→  const isPhotographing = !photoshoot && currentlyActive !== null
   125→

   127→  const sentStudents = useMemo<Student[]>(() => {
   128→    if (!photoshoot) return []
   129→    return channelStudents.filter((s) => s.status === 'sent')

   134→    const proj = currentProjectRef.current
   135→    if (!proj) return {}
   136→    const result: Record<number, boolean> = {}

   139→      result[ch] = proj.photoHistory.some((h) => h.student.id === studentId && h.channel === ch)

   147→    const q = searchQuery.toLowerCase().trim()
   148→    return channelStudents.filter(
   149→      (s) =>

   172→      if (data.channel !== myChannelRef.current) return

   198→          proj.photoHistory ?? [],
   199→          curProj.photoHistory,

   202→          ...curProj,
   203→          database: mergedDb,
   204→          photoHistory: mergedPhotoHistory,
   205→          config: mergedConfig,

   210→    onLocal('SYNC_DB', handleSyncDb)
   211→    return () => { offLocal('SYNC_DB', handleSyncDb) }

   220→
   221→      // Non-photoshoot: already handled by STUDENT_DONE event — skip here
   222→      if (!photoshoot) return
   223→

   230→        photos: data.photos,
   231→        channel: data.channel,
   232→      }

   234→      const existing = curProj.photoHistory.findIndex(
   235→        (h) => h.student.id === data.student.id && h.channel === data.channel,

   242→        newHistory = [...curProj.photoHistory, historyItem]

   245→      // Check if all required channels have completed
   246→      const chCount = channelCount(curProj.config.mode)
   247→      let allChannelsDone = true
   248→      for (let ch = 1; ch <= chCount; ch++) {
   249→        const chDone = newHistory.some((h) => h.student.id === data.student.id && h.channel === ch)
   250→        if (!chDone) {
   251→          allChannelsDone = false

   254→      }
   255→
   256→      const updatedProject = {
   257→        ...curProj,
   258→        database: curProj.database.map((s) =>
   259→          s.id === data.student.id && allChannelsDone ? { ...s, status: 'done' as StudentStatus } : s
   260→        ),
   261→        photoHistory: newHistory,
   262→      }
   263→      updateCurrentProject(updatedProject)
   264→      saveProjectsToStorageNow()
   265→
   266→      console.log('[SAATIRIL MC] PHOTOS_SAVED: allChannelsDone =', allChannelsDone, 'for', data.student.nama)
   267→    }
   268→
   269→    onLocal('PHOTOS_SAVED', handlePhotosSaved)
   270→    return () => { offLocal('PHOTOS_SAVED', handlePhotosSaved) }
   271→  }, [updateStudentStatus, updateCurrentProject, saveProjectsToStorageNow, photoshoot])
   272→
   273→  // ── Socket: OP_PROGRESS
   274→  useEffect(() => {
   275→    const handleOpProgress = (data: OpProgressData) => {
   276→      if (!photoshoot && data.channel !== myChannelRef.current) return

   279→      setOpProgressChannel(data.channel)
   280→    }
   281→

   283→    return () => { offLocal('OP_PROGRESS', handleOpProgress) }

   285→
   286→  // ── Socket: MC_CALL
   287→  useEffect(() => {
   288→    const handleMcCall = (data: { student: Student; channel: number }) => {

   290→      updateStudentStatus(data.student.id, data.student.status)
   291→    }
   292→
   293→    onLocal('MC_CALL', handleMcCall)
   294→    return () => { offLocal('MC_CALL', handleMcCall) }
   295→  }, [updateStudentStatus, photoshoot])
   296→

   300→

   305→    const latestProject = useSaatirilStore.getState().currentProject
   306→    if (!latestProject) return
   307→

   309→      ...latestProject,
   310→      database: latestProject.database.map((s) =>
   311→        s.id === nextPending.id ? { ...s, status: newStatus } : s
   312→      ),
   313→    }

   315→    setOpProgressText('')
   316→
   317→    // PRIORITY: MC_CALL first (lightweight, operator gets student immediately)
   318→    emitLocal('MC_CALL', {
   319→      student: { ...nextPending, status: newStatus },
   320→      channel: myChannel,
   321→    })
   322→    // Then SYNC_DB for consistency (now lightweight — photos stripped)
   323→    emitLocal('SYNC_DB', { project: stripFrameForSync(updatedProject) })
   324→  }, [
   325→    nextPending,
   326→    currentProject,
   327→    myChannel,
   328→    updateStudentStatus,
   329→    updateCurrentProject,
   330→    saveProjectsToStorageNow,
   331→  ])
   332→
   333→  // ── Photoshoot: send selected student to operator(s)
   334→  const handleSendToOperator = useCallback(() => {
   335→    if (!selectedStudent || !currentProject) return
   336→
   337→    if (dualPhotoshoot) {
   338→      // Send to BOTH channels
   339→      const newStatus: StudentStatus = 'sent'
   340→
   341→      const latestProject = useSaatirilStore.getState().currentProject
   342→      if (!latestProject) return
   343→
   344→      // Update student status to 'sent' in database
   345→      const updatedProject = {
   346→        ...latestProject,
   347→        database: latestProject.database.map((s) =>
   348→          s.id === selectedStudent.id ? { ...s, status: newStatus } : s
   349→        ),
   350→      }
   351→
   352→      updateStudentStatus(selectedStudent.id, newStatus)
   353→      updateCurrentProject(updatedProject)
   354→      saveProjectsToStorageNow()
   355→
   356→      // Send MC_CALL to both channels FIRST (operators get student immediately)
   357→      emitLocal('MC_CALL', {
   358→        student: { ...selectedStudent, status: newStatus, assignedChannel: 1 },
   359→        channel: 1,
   360→      })
   361→      emitLocal('MC_CALL', {
   362→        student: { ...selectedStudent, status: newStatus, assignedChannel: 2 },
   363→        channel: 2,
   364→      })
   365→      // Then SYNC_DB for consistency (now lightweight — photos stripped)

   373→

   375→        ...latestProject,
   376→        database: latestProject.database.map((s) =>
   377→          s.id === selectedStudent.id ? { ...s, status: newStatus } : s

   379→      }

   381→      updateStudentStatus(selectedStudent.id, newStatus)
   382→      updateCurrentProject(updatedProject)

   386→        student: { ...selectedStudent, status: newStatus },
   387→        channel: 1,

   389→      emitLocal('SYNC_DB', { project: stripFrameForSync(updatedProject) })
   390→    }

   399→  // their buffer / active target / photoHistory. This is necessary because
   400→  // the normal SYNC_DB merge (mergeDatabases) BLOCKS status regression
   401→  // (pending priority 0 < sent/done), and preservePhotoHistoryOnSync does
   402→  // not propagate photoHistory deletions — so a reset via SYNC_DB alone would
   403→  // be silently dropped on the receiver side.
   404→  const handleResetForRetake = useCallback((student: Student) => {
   405→    const latestProject = useSaatirilStore.getState().currentProject
   406→    if (!latestProject) return
   407→
   408→    // Remove ALL photoHistory entries for this student (every channel) so the
   409→    // operator queue (which filters via `alreadyPhotographed`) will re-show
   410→    // the student after re-send.
   411→    const cleanedPhotoHistory = latestProject.photoHistory.filter(
   412→      (h) => h.student.id !== student.id,
   413→    )
   414→
   415→    const updatedProject: typeof latestProject = {
   416→      ...latestProject,
   417→      database: latestProject.database.map((s) =>
   418→        s.id === student.id ? { ...s, status: 'pending' as StudentStatus } : s,
   419→      ),
   420→      photoHistory: cleanedPhotoHistory,
   421→    }
   422→
   423→    updateStudentStatus(student.id, 'pending')
   424→    updateCurrentProject(updatedProject)
   425→    saveProjectsToStorageNow()
   426→
   427→    // Emit STUDENT_RESET to every relevant channel — this bypasses the merge
   428→    // priority logic and tells each operator to: clear mcCallBuffer entry,
   429→    // clear opCurrentTarget if it matches, remove their photoHistory entry,
   430→    // and set the student status to 'pending' locally.
   431→    const channels = dualPhotoshoot ? [1, 2] : [1]
   432→    for (const ch of channels) {
   433→      emitLocal('STUDENT_RESET', { studentId: student.id, channel: ch })
   434→    }
   435→    // Also emit SYNC_DB for consistency (photoHistory + status now cleaned)
   436→    emitLocal('SYNC_DB', { project: stripFrameForSync(updatedProject) })
   437→
   438→    // Pre-select the student for easy re-send
   439→    setSelectedStudent({ ...student, status: 'pending' })
   440→    setSearchQuery(student.nama)
   441→  }, [dualPhotoshoot, updateStudentStatus, updateCurrentProject, saveProjectsToStorageNow])
   442→
   443→  // ── Render helpers

   470→          style={{

   472→            color: selectedStudent && selectedStudent.status !== 'done' ? THEME.bg : THEME.muted,
   473→            border: `2px solid ${selectedStudent && selectedStudent.status !== 'done' ? THEME.emerald : THEME.border}`,
   474→            boxShadow: selectedStudent && selectedStudent.status !== 'done' ? `0 0 20px ${THEME.emerald}44` : 'none',
   475→          }}
   476→        >
   477→          <Send className="size-5" />
   478→          {dualPhotoshoot ? 'KIRIM KE 2 KAMERA' : 'KIRIM KE OPERATOR'}

   480→      )

   482→
   483→    // Non-photoshoot: existing blocking flow

   488→            disabled

   490→            style={{

   508→          style={{
   509→            backgroundColor: THEME.gold,

   521→    return (
   522→      <Button
   523→        disabled
   524→        className={`w-full font-bold cursor-not-allowed ${isMobile ? 'h-12 text-sm' : 'h-14 text-lg'}`}
   525→        style={{
   526→          backgroundColor: THEME.panel,
   527→          color: THEME.muted,
