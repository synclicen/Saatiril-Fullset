    66→// ─── Component ─────────────────────────────────────────────────────────────────
    67→export default function ProjectSetup() {
    68→  const { toast } = useToast()
    69→  const {
    70→    addProject,
    71→    setCurrentProject,
    72→    setCurrentScreen,
    73→    saveProjectsToStorageNow,
    74→  } = useSaatirilStore()
    75→
    76→  // ── Form state ─────────────────────────────────────────────────────────────
    77→  const [projectName, setProjectName] = useState('')
    78→  const [cameraMode, setCameraMode] = useState<CameraMode>('single')
    79→  const [ratio, setRatio] = useState('4:3')
    80→  const [preset, setPreset] = useState('original')
    81→  const [targetFolder, setTargetFolder] = useState('C:\\SAATIRIL_System_Out')
    82→  const [frameData, setFrameData] = useState<string | null>(null)
    83→  const [frameFileName, setFrameFileName] = useState<string>('')
    84→
    85→  // Excel data: single mode uses index 0; dual uses 0 (kiri) & 1 (kanan)
    86→  const [excelData, setExcelData] = useState<[ParsedExcelData | null, ParsedExcelData | null]>([null, null])
    87→
    88→  // Loading states for file parsing
    89→  const [parsingExcel, setParsingExcel] = useState<[boolean, boolean]>([false, false])
    90→  const [parsingFrame, setParsingFrame] = useState(false)
    91→
    92→  // Refs for hidden file inputs
    93→  const excelInputRef0 = useRef<HTMLInputElement>(null)
    94→  const excelInputRef1 = useRef<HTMLInputElement>(null)
    95→  const frameInputRef = useRef<HTMLInputElement>(null)
    96→
    97→  // ── Excel parsing ──────────────────────────────────────────────────────────
    98→  const parseExcelFile = useCallback(
    99→    (file: File, channelIndex: number) => {
   100→      setParsingExcel((prev) => {
   101→        const next = [...prev] as [boolean, boolean]
   102→        next[channelIndex] = true
   103→        return next
   104→      })
   105→
   106→      const reader = new FileReader()
   107→      reader.onload = (e) => {
   108→        try {
   109→          const data = new Uint8Array(e.target?.result as ArrayBuffer)
   110→          const workbook = XLSX.read(data, { type: 'array' })
   111→          const firstSheet = workbook.Sheets[workbook.SheetNames[0]]
   112→          const jsonData = XLSX.utils.sheet_to_json<Record<string, unknown>>(firstSheet, {
   113→            defval: '',
   114→          })
   115→
   116→          if (jsonData.length === 0) {
   117→            toast({
   118→              title: 'File Kosong',
   119→              description: 'File Excel tidak memiliki data.',
   120→              variant: 'destructive',
   121→            })
   122→            setParsingExcel((prev) => {
   123→              const next = [...prev] as [boolean, boolean]
   124→              next[channelIndex] = false
   125→              return next
   126→            })
   127→            return
   128→          }
   129→
   130→          // Identify NIM and Nama columns
   131→          const headers = Object.keys(jsonData[0])
   132→          const nimCol = headers.find((h) => {
   133→            const lower = h.toLowerCase().trim()
   134→            return lower.includes('nim') || lower.includes('nis') || lower.includes('id')
   135→          })
   136→          const namaCol = headers.find((h) => {
   137→            const lower = h.toLowerCase().trim()
   138→            return lower.includes('nama') || lower.includes('name')
   139→          })
   140→
   141→          if (!nimCol || !namaCol) {
   142→            toast({
   143→              title: 'Kolom Tidak Ditemukan',
   144→              description: `Tidak dapat menemukan kolom NIM/ID dan Nama. Kolom yang ditemukan: ${headers.join(', ')}`,
   145→              variant: 'destructive',
   146→            })
   147→            setParsingExcel((prev) => {
   148→              const next = [...prev] as [boolean, boolean]
   149→              next[channelIndex] = false
   150→              return next
   151→            })
   152→            return
   153→          }
   154→
   155→          const channelNumber = channelIndex + 1
   156→          const students: Student[] = jsonData
   157→            .filter((row) => {
   158→              const nim = String(row[nimCol] ?? '').trim()
   159→              const nama = String(row[namaCol] ?? '').trim()
   160→              return nim !== '' && nama !== ''
   161→            })
   162→            .map((row) => ({
   163→              id: `ID_${Date.now()}_${Math.random().toString(36).substring(2, 9)}`,
   164→              nim: String(row[nimCol] ?? '').trim(),
   165→              nama: String(row[namaCol] ?? '').trim(),
   166→              status: 'pending' as const,
   167→              assignedChannel: channelNumber,
   168→            }))
   169→
   170→          setExcelData((prev) => {
   171→            const next = [...prev] as [ParsedExcelData | null, ParsedExcelData | null]
   172→            next[channelIndex] = { fileName: file.name, students }
   173→            return next
   174→          })
   175→
   176→          toast({
   177→            title: 'Data Berhasil Dimuat',
   178→            description: `${students.length} peserta terbaca dari "${file.name}"`,
   179→          })
   180→        } catch (err) {
   181→          console.error('[SAATIRIL] Excel parse error:', err)
   182→          toast({
   183→            title: 'Gagal Membaca File',
   184→            description: 'Pastikan file berformat .xlsx atau .xls yang valid.',
   185→            variant: 'destructive',
   186→          })
   187→        } finally {
   188→          setParsingExcel((prev) => {
   189→            const next = [...prev] as [boolean, boolean]
   190→            next[channelIndex] = false
   191→            return next
   192→          })
   193→        }
   194→      }
   195→      reader.onerror = () => {
   196→        toast({
   197→          title: 'Gagal Membaca File',
   198→          description: 'Terjadi kesalahan saat membaca file.',
   199→          variant: 'destructive',
   200→        })
   201→        setParsingExcel((prev) => {
   202→          const next = [...prev] as [boolean, boolean]
   203→          next[channelIndex] = false
   204→          return next
   205→        })
   206→      }
   207→      reader.readAsArrayBuffer(file)
   208→    },
   209→    [toast],
   210→  )
   211→
   212→  // ── Frame overlay parsing ──────────────────────────────────────────────────
   213→  const parseFrameFile = useCallback(
   214→    (file: File) => {
   215→      if (!file.name.toLowerCase().endsWith('.png')) {
   216→        toast({
   217→          title: 'Format Salah',
   218→          description: 'Frame overlay harus berformat .PNG (transparan).',
   219→          variant: 'destructive',
   220→        })
   221→        return
   222→      }
   223→
   224→      setParsingFrame(true)
   225→      const reader = new FileReader()
   226→      reader.onload = (e) => {
   227→        const dataUrl = e.target?.result as string
   228→        setFrameData(dataUrl)
   229→        setFrameFileName(file.name)
   230→        setParsingFrame(false)
   231→        toast({
   232→          title: 'Frame Dimuat',
   233→          description: `"${file.name}" berhasil dimuat sebagai overlay.`,
   234→        })
   235→      }
   236→      reader.onerror = () => {
   237→        toast({
   238→          title: 'Gagal Membaca Frame',
   239→          description: 'Terjadi kesalahan saat membaca file frame.',
   240→          variant: 'destructive',
   241→        })
   242→        setParsingFrame(false)
   243→      }
   244→      reader.readAsDataURL(file)
   245→    },
   246→    [toast],
   247→  )
   248→
   249→  // ── Drop zone handlers ─────────────────────────────────────────────────────
   250→  const handleExcelDrop = useCallback(
   251→    (e: React.DragEvent, channelIndex: number) => {
   252→      e.preventDefault()
   253→      e.stopPropagation()
   254→      const file = e.dataTransfer.files[0]
   255→      if (file) parseExcelFile(file, channelIndex)
   256→    },
   257→    [parseExcelFile],
   258→  )
   259→
   260→  const handleFrameDrop = useCallback(
   261→    (e: React.DragEvent) => {
   262→      e.preventDefault()
   263→      e.stopPropagation()
   264→      const file = e.dataTransfer.files[0]
   265→      if (file) parseFrameFile(file)
   266→    },
   267→    [parseFrameFile],
   268→  )
   269→
   270→  const handleDragOver = useCallback((e: React.DragEvent) => {
   271→    e.preventDefault()
   272→    e.stopPropagation()
   273→  }, [])
   274→
   275→  // ── Form validation ────────────────────────────────────────────────────────
   276→  const isNameValid = projectName.trim().length > 0
   277→  const isSingleDataReady = (cameraMode === 'single' || cameraMode === 'single-photoshoot' || cameraMode === 'dual-photoshoot') && excelData[0] !== null && excelData[0]!.students.length > 0
   278→  const isDualDataReady =
   279→    cameraMode === 'dual' &&
   280→    excelData[0] !== null &&
   281→    excelData[0]!.students.length > 0 &&
   282→    excelData[1] !== null &&
   283→    excelData[1]!.students.length > 0
   284→  const isDataReady = isSingleDataReady || isDualDataReady
   285→  const canStart = isNameValid && isDataReady
   286→
   287→  // ── Submit / create project ────────────────────────────────────────────────
   288→  const handleCreateProject = useCallback(() => {
   289→    if (!canStart) return
   290→
   291→    const allStudents: Student[] = []
   292→
   293→    if (cameraMode === 'dual') {
   294→      allStudents.push(...(excelData[0]?.students ?? []))
   295→      allStudents.push(...(excelData[1]?.students ?? []))
   296→    } else {
   297→      // single, single-photoshoot, dual-photoshoot: only 1 data upload
   298→      allStudents.push(...(excelData[0]?.students ?? []))
   299→    }
   300→
   301→    // Re-generate unique IDs to avoid collision from parallel parsing
   302→    const finalStudents: Student[] = allStudents.map((s, i) => ({
   303→      ...s,
   304→      id: `ID_${Date.now()}_${i}_${Math.random().toString(36).substring(2, 9)}`,
   305→    }))
   306→
   307→    const projectId = `PRJ_${Date.now()}_${Math.random().toString(36).substring(2, 9)}`
   308→
   309→    // Create subfolder based on project name under target folder
   310→    const sanitizedName = projectName.trim().replace(/[^a-zA-Z0-9_\-\s]/g, '').replace(/\s+/g, '_')
   311→    const finalTargetFolder = targetFolder
   312→      ? `${targetFolder.replace(/[\\/]+$/, '')}\\${sanitizedName}`
   313→      : `C:\\SAATIRIL_System_Out\\${sanitizedName}`
   314→
   315→    const project = {
   316→      id: projectId,
   317→      name: projectName.trim(),
   318→      config: {
   319→        mode: cameraMode,
   320→        ratio,
   321→        preset,
   322→        targetFolder: finalTargetFolder,
   323→        frame: frameData,
   324→      },
   325→      database: finalStudents,
   326→      photoHistory: [],
   327→    }
   328→
   329→    addProject(project)
   330→    setCurrentProject(project)
   331→
   332→    // IMMEDIATE save before navigation to prevent data loss on first launch
   333→    saveProjectsToStorageNow()
   334→
   335→    // ── Create project folder on disk (Electron only) ────────────────────
   336→    const api = window.saatirilAPI
   337→    if (api?.createFolder) {
   338→      api.createFolder(finalTargetFolder).then((result: { success: boolean; path?: string; error?: string }) => {
   339→        if (result.success) {
   340→          console.log(`[SAATIRIL] ✅ Project folder created: ${result.path}`)
   341→          toast({
   342→            title: 'Proyek Dibuat!',
   343→            description: `"${projectName.trim()}" — ${finalStudents.length} peserta dimuat. Folder: ${finalTargetFolder}`,
   344→          })
   345→        } else {
   346→          console.error('[SAATIRIL] Failed to create project folder:', result.error)
   347→          toast({
   348→            title: 'Proyek Dibuat!',
   349→            description: `"${projectName.trim()}" — ${finalStudents.length} peserta. ⚠️ Folder gagal: ${result.error}`,
   350→            variant: 'destructive',
   351→          })
   352→        }
   353→      }).catch((err: Error) => {
   354→        console.error('[SAATIRIL] createFolder IPC error:', err)
   355→        toast({
   356→          title: 'Proyek Dibuat!',
   357→          description: `"${projectName.trim()}" — ${finalStudents.length} peserta dimuat.`,
   358→        })
   359→      })
   360→    } else {
   361→      toast({
   362→        title: 'Proyek Dibuat!',
   363→        description: `"${projectName.trim()}" — ${finalStudents.length} peserta dimuat.`,
   364→      })
   365→    }
   366→
   367→    // Sync database over LAN — strip frame data to save bandwidth
   368→    emitLocal('SYNC_DB', { project: stripFrameForSync(project) })
   369→
   370→    // Small delay to ensure state is persisted before navigation
   371→    setTimeout(() => {
   372→      setCurrentScreen('app')
   373→    }, 50)
   374→  }, [
   375→    canStart,
   376→    cameraMode,
   377→    excelData,
   378→    projectName,
   379→    ratio,
   380→    preset,
   381→    targetFolder,
   382→    frameData,
   383→    addProject,
   384→    setCurrentProject,
   385→    saveProjectsToStorageNow,
   386→    setCurrentScreen,
   387→    toast,
   388→  ])
   389→
   390→  // ── Upload zone component ──────────────────────────────────────────────────
   391→  const renderUploadZone = (
   392→    channelIndex: number,
   393→    label: string,
   394→    accentColor: string,
   395→    accentBorder: string,
   396→  ) => {
   397→    const data = excelData[channelIndex]
   398→    const isLoading = parsingExcel[channelIndex]
   399→    const inputRef = channelIndex === 0 ? excelInputRef0 : excelInputRef1
   400→
   401→    return (
   402→      <div className="space-y-2">
   403→        <Label className="text-sm font-medium" style={{ color: accentColor }}>
   404→          {label}
   405→        </Label>
   406→        <div
   407→          onDrop={(e) => handleExcelDrop(e, channelIndex)}
   408→          onDragOver={handleDragOver}
   409→          onClick={() => inputRef.current?.click()}
   410→          className={`
   411→            relative cursor-pointer rounded-lg border-2 border-dashed p-6
   412→            transition-all duration-200
   413→            hover:border-opacity-100 hover:bg-white/5
   414→            ${data ? accentBorder : 'border-[#533485]'}
   415→          `}
   416→        >
   417→          <input
   418→            ref={inputRef}
   419→            type="file"
   420→            accept=".xlsx,.xls,.csv"
   421→            className="hidden"
   422→            onChange={(e) => {
   423→              const file = e.target.files?.[0]
   424→              if (file) parseExcelFile(file, channelIndex)
   425→              // Reset so same file can be re-selected
   426→              e.target.value = ''
   427→            }}
   428→          />
   429→
   430→          <div className="flex flex-col items-center gap-2 text-center">
   431→            {isLoading ? (
   432→              <>
   433→                <div className="h-8 w-8 animate-spin rounded-full border-2 border-[#533485] border-t-[#d4af37]" />
   434→                <p className="text-xs text-[#c4b5fd]">Membaca file...</p>
   435→              </>
   436→            ) : data ? (
   437→              <>
   438→                <FileSpreadsheet className="h-8 w-8" style={{ color: accentColor }} />
   439→                <p className="text-sm font-medium" style={{ color: accentColor }}>
   440→                  Data siap: {data.students.length} Peserta Terbaca
   441→                </p>
   442→                <p className="text-xs text-[#c4b5fd]">{data.fileName}</p>
   443→                <button
   444→                  onClick={(e) => {
   445→                    e.stopPropagation()
   446→                    setExcelData((prev) => {
   447→                      const next = [...prev] as [ParsedExcelData | null, ParsedExcelData | null]
   448→                      next[channelIndex] = null
   449→                      return next
   450→                    })
   451→                  }}
   452→                  className="mt-1 rounded-full p-1 text-[#c4b5fd] transition-colors hover:bg-white/10 hover:text-red-400"
   453→                >
   454→                  <X className="h-4 w-4" />
   455→                </button>
   456→              </>
   457→            ) : (
   458→              <>
   459→                <Upload className="h-8 w-8 text-[#533485]" />
   460→                <p className="text-sm text-[#c4b5fd]">
   461→                  Drag & drop file Excel di sini
   462→                </p>
   463→                <p className="text-xs text-[#533485]">.xlsx / .xls / .csv</p>
   464→              </>
   465→            )}
   466→          </div>
   467→        </div>
   468→      </div>
   469→    )
   470→  }
   471→
   472→  // ── Render ─────────────────────────────────────────────────────────────────
   473→  return (
   474→    <div className="relative h-screen bg-[#1a0b2e] backdrop-blur-sm overflow-hidden">
   475→      {/* Subtle gradient overlay */}
   476→      <div className="pointer-events-none absolute inset-0 bg-gradient-to-br from-[#1a0b2e] via-[#2a164a]/30 to-[#1a0b2e]" />
   477→
   478→      <div className="relative z-10 flex h-full flex-col">
   479→        {/* ── Header ─────────────────────────────────────────────────────────── */}
   480→        <header className="flex items-center gap-3 border-b border-[#533485]/50 px-5 py-2.5">
   481→          <Button
   482→            variant="ghost"
   483→            size="icon"
   484→            onClick={() => setCurrentScreen('hub')}
   485→            className="text-[#c4b5fd] hover:bg-white/10 hover:text-[#d4af37]"
   486→          >
   487→            <ArrowLeft className="h-4 w-4" />
   488→          </Button>
   489→          <div>
   490→            <h1 className="text-base font-bold text-white">
   491→              <span className="text-[#d4af37]">SAATIRIL</span> — Buat Proyek Baru
   492→            </h1>
   493→          </div>
   494→        </header>
   495→
   496→        {/* ── Main Content: 2-Column Layout ──────────────────────────────────── */}
   497→        <main className="flex-1 overflow-y-auto p-4">
   498→          <div className="mx-auto grid max-w-6xl gap-4 lg:grid-cols-2">
   499→            {/* ── LEFT COLUMN ──────────────────────────────────────────────── */}
   500→            <div className="flex flex-col gap-3">
   501→              {/* Project Name */}
   502→              <Card className="border-[#533485] bg-[#2a164a] shadow-lg">
   503→                <CardHeader className="pb-2">
   504→                  <CardTitle className="flex items-center gap-2 text-base text-white">
   505→                    <Monitor className="h-5 w-5 text-[#d4af37]" />
   506→                    Nama Proyek
   507→                  </CardTitle>
   508→                </CardHeader>
   509→                <CardContent>
   510→                  <Input
   511→                    placeholder="Masukkan nama event / proyek..."
   512→                    value={projectName}
   513→                    onChange={(e) => setProjectName(e.target.value)}
   514→                    className="border-[#533485] bg-[#3b2263] text-white placeholder:text-[#533485] focus-visible:border-[#d4af37] focus-visible:ring-[#d4af37]/30"
   515→                  />
   516→                </CardContent>
   517→              </Card>
   518→
   519→              {/* Camera Mode */}
   520→              <Card className="border-[#533485] bg-[#2a164a] shadow-lg">
   521→                <CardHeader className="pb-2">
   522→                  <CardTitle className="flex items-center gap-2 text-base text-white">
   523→                    <Camera className="h-5 w-5 text-[#d4af37]" />
   524→                    Mode Kamera
   525→                  </CardTitle>
   526→                </CardHeader>
   527→                <CardContent className="space-y-3">
   528→                  <Select
   529→                    value={cameraMode}
   530→                    onValueChange={(val) => {
   531→                      setCameraMode(val as CameraMode)
   532→                      // Reset kanan data when switching away from dual
   533→                      if (val !== 'dual') {
   534→                        setExcelData((prev) => [prev[0], null])
   535→                      }
   536→                    }}
