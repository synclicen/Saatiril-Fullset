   255→   255→      socket.off('auth-success', handleAuthSuccess)

   287→   287→          const mergedConfig = preserveFrameOnSync(data.project.config, curProj.config)

   294→   294→            database: mergedDb,
   295→   295→            photoHistory: mergedPhotoHistory,
   296→   296→            config: mergedConfig,
   297→   297→          })
   298→   298→        } else {
   299→   299→          updateCurrentProject(data.project)
   300→   300→        }
   301→   301→      } else if (role === 'admin' && data.project) {
   302→   302→        // For admin: merge database with incoming (prevents channel data overwrite in dual mode)
   303→   303→        if (curProj && data.project.id === curProj.id) {
   304→   304→          const mergedDb = mergeDatabases(curProj.database, data.project.database)
   305→   305→          const mergedConfig = preserveFrameOnSync(data.project.config, curProj.config)
   306→   306→          const mergedPhotoHistory = preservePhotoHistoryOnSync(
   307→   307→            data.project.photoHistory ?? [],
   308→   308→            curProj.photoHistory,
   309→   309→          )
   310→   310→          updateCurrentProject({
   311→   311→            ...curProj,
   312→   312→            database: mergedDb,
   313→   313→            photoHistory: mergedPhotoHistory,
   314→   314→            config: mergedConfig,

   380→   380→    store.setCurrentTab('admin')
   381→   381→    store.setMyChannel(1)
   382→   382→    store.resetOpState()
   383→   383→    setCurrentScreen('hub')
   384→   384→  }, [setCurrentScreen])
   385→   385→
   386→   386→  const handleTabChange = useCallback(
   387→   387→    (tab: AppTab) => {
   388→   388→      if (myRole === 'admin') {
   389→   389→        setCurrentTab(tab)
   390→   390→      }
   391→   391→    },
   392→   392→    [myRole, setCurrentTab],
   393→   393→  )
   394→   394→
   395→   395→  const handleChannelSelect = useCallback(
   396→   396→    (channel: string) => {

   400→   400→  )

   408→   408→  // Server-side validation: show prompt when server says password is required
   409→   409→  // OR when we have a __PASSWORD_SET__ marker and haven't authenticated yet
   410→   410→  const needsPassword = myRole !== 'admin' && !sessionPasswordVerified && (
   411→   411→    serverRequiresPassword ||
   412→   412→    (currentProject?.config?.sessionPassword && currentProject.config.sessionPassword !== '__PASSWORD_SET__') ||
   413→   413→    (currentProject?.config?.sessionPassword === '__PASSWORD_SET__' && !isSocketAuthenticated())
   414→   414→  )
   415→   415→  if (needsPassword) {
   416→   416→    return (
   417→   417→      <div
   418→   418→        className="flex h-dvh flex-col items-center justify-center gap-6 px-6"
   419→   419→        style={{ backgroundColor: THEME.bg }}
   420→   420→      >
   421→   421→        <div
   422→   422→          className="flex size-20 items-center justify-center rounded-full"
   423→   423→          style={{ backgroundColor: `${THEME.gold}15`, borderWidth: 1, borderColor: `${THEME.gold}33` }}
   424→   424→        >
   425→   425→          <Lock className="size-10" style={{ color: THEME.gold }} />
   426→   426→        </div>
   427→   427→        <div className="text-center">
   428→   428→          <h2 className="text-xl font-bold text-white">Password Sesi Diperlukan</h2>
   429→   429→          <p className="mt-2 text-sm" style={{ color: THEME.muted }}>
   430→   430→            Admin telah mengatur password untuk sesi ini.
   431→   431→            Masukkan password untuk bergabung.
   432→   432→          </p>
   433→   433→        </div>
   434→   434→        <div className="w-full max-w-xs space-y-3">
   435→   435→          <Input
   436→   436→            type="password"
   437→   437→            placeholder="Masukkan password sesi..."
   438→   438→            value={sessionPasswordInput}
   439→   439→            onChange={(e) => { setSessionPasswordInput(e.target.value); setSessionPasswordError(false) }}
   440→   440→            onKeyDown={(e) => {
   441→   441→              if (e.key === 'Enter' && sessionPasswordInput.trim()) {
   442→   442→                reidentifyWithPassword(sessionPasswordInput.trim())
   443→   443→              }
   444→   444→            }}
   445→   445→            className="h-10 text-center font-mono text-sm"
   446→   446→            style={{
