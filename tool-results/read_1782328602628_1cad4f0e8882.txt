    32→    32→function getActiveChannel(status: StudentStatus): number | null {
    33→    33→  if (!isActiveStatus(status)) return null
    34→    34→  const ch = status.split('_')[1]
    35→    35→  return ch ? parseInt(ch, 10) : null
    36→    36→}

    38→    38→function statusLabel(status: StudentStatus): string {
    39→    39→  if (status === 'pending') return 'Menunggu'
    40→    40→  if (status === 'sent') return 'Dikirim'
    41→    41→  if (status === 'done') return 'Selesai'
    42→    42→  const ch = getActiveChannel(status)
    43→    43→  return ch != null ? `Foto Ch.${ch}` : 'Aktif'
    44→    44→}

   396→   396→  // ── Photoshoot: reset (for retake)
   397→   397→  // Clears the student's photoHistory + resets status to 'pending' + emits a
   398→   398→  // dedicated STUDENT_RESET event so operators (and admin) explicitly clear
   399→   399→  // their buffer / active target / photoHistory. This is necessary because
   400→   400→  // the normal SYNC_DB merge (mergeDatabases) BLOCKS status regression
   401→   401→  // (pending priority 0 < sent/done), and preservePhotoHistoryOnSync does
   402→   402→  // not propagate photoHistory deletions — so a reset via SYNC_DB alone would
   403→   403→  // be silently dropped on the receiver side.
   404→   404→  const handleResetForRetake = useCallback((student: Student) => {
   405→   405→    const latestProject = useSaatirilStore.getState().currentProject
   406→   406→    if (!latestProject) return
   407→   407→
   408→   408→    // Remove ALL photoHistory entries for this student (every channel) so the
   409→   409→    // operator queue (which filters via `alreadyPhotographed`) will re-show
   410→   410→    // the student after re-send.
   411→   411→    const cleanedPhotoHistory = latestProject.photoHistory.filter(
   412→   412→      (h) => h.student.id !== student.id,
   413→   413→    )
   414→   414→
   415→   415→    const updatedProject: typeof latestProject = {
   416→   416→      ...latestProject,
   417→   417→      database: latestProject.database.map((s) =>
   418→   418→        s.id === student.id ? { ...s, status: 'pending' as StudentStatus } : s,
   419→   419→      ),
   420→   420→      photoHistory: cleanedPhotoHistory,
   421→   421→    }
   422→   422→
   423→   423→    updateStudentStatus(student.id, 'pending')
   424→   424→    updateCurrentProject(updatedProject)
   425→   425→    saveProjectsToStorageNow()
   426→   426→
   427→   427→    // Emit STUDENT_RESET to every relevant channel — this bypasses the merge
   428→   428→    // priority logic and tells each operator to: clear mcCallBuffer entry,
   429→   429→    // clear opCurrentTarget if it matches, remove their photoHistory entry,
   430→   430→    // and set the student status to 'pending' locally.
   431→   431→    const channels = dualPhotoshoot ? [1, 2] : [1]
   432→   432→    for (const ch of channels) {
   433→   433→      emitLocal('STUDENT_RESET', { studentId: student.id, channel: ch })
   434→   434→    }
   435→   435→    // Also emit SYNC_DB for consistency (photoHistory + status now cleaned)
   436→   436→    emitLocal('SYNC_DB', { project: stripFrameForSync(updatedProject) })
   437→   437→
   438→   438→    // Pre-select the student for easy re-send
   439→   439→    setSelectedStudent({ ...student, status: 'pending' })
   440→   440→    setSearchQuery(student.nama)
   441→   441→  }, [dualPhotoshoot, updateStudentStatus, updateCurrentProject, saveProjectsToStorageNow])
   442→   442→
   443→   443→  // ── Render helpers
   444→   444→  const renderCallButton = () => {
