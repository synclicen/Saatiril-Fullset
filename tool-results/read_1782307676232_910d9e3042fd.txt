   113→// ─── Ratio parser ───────────────────────────────────────────────────────────
   114→function parseRatio(ratioStr: string): number {
   115→  const parts = ratioStr.split(':')
   116→  if (parts.length === 2) {
   117→    const w = parseFloat(parts[0])
   118→    const h = parseFloat(parts[1])
   119→    if (w > 0 && h > 0) return w / h
   120→  }
   121→  return 4 / 3
   122→}

   157→function fitAspectRatio(
   158→  availW: number,
   159→  availH: number,
   160→  ratio: number,
   161→): { width: number; height: number } {
   162→  const hFromW = availW / ratio
   163→  if (hFromW <= availH) {
   164→    return { width: availW, height: hFromW }
   165→  }
   166→  const wFromH = availH * ratio
   167→  return { width: wFromH, height: availH }
   168→}
