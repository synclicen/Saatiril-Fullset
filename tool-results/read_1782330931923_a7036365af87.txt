   109→// ─── Ratio parser ───────────────────────────────────────────────────────────
   110→function parseRatio(ratioStr: string): number {
   111→  const parts = ratioStr.split(':')
   112→  if (parts.length === 2) {
   113→    const w = parseFloat(parts[0])
   114→    const h = parseFloat(parts[1])
   115→    if (w > 0 && h > 0) return w / h
   116→  }
   117→  return 4 / 3
   118→}

   153→function fitAspectRatio(
   154→  availW: number,
   155→  availH: number,
   156→  ratio: number,
   157→): { width: number; height: number } {
   158→  const hFromW = availW / ratio
   159→  if (hFromW <= availH) {
   160→    return { width: availW, height: hFromW }
   161→  }
   162→  const wFromH = availH * ratio
   163→  return { width: wFromH, height: availH }
   164→}
