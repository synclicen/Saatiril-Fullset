    28→    28→function isActiveStatus(status: StudentStatus): boolean {
    29→    29→  return status.startsWith('active')
    30→    30→}
    31→    31→
    32→    32→function getActiveChannel(status: StudentStatus): number | null {
    33→    33→  if (!isActiveStatus(status)) return null
    34→    34→  const ch = status.split('_')[1]
    35→    35→  return ch ? parseInt(ch, 10) : null
    36→    36→}
    37→    37→
    38→    38→function statusLabel(status: StudentStatus): string {
    39→    39→  if (status === 'pending') return 'Menunggu'
    40→    40→  if (status === 'sent') return 'Dikirim'
    41→    41→  if (status === 'done') return 'Selesai'
    42→    42→  const ch = getActiveChannel(status)
    43→    43→  return ch != null ? `Foto Ch.${ch}` : 'Aktif'
    44→    44→}

    84→    84→  const [opProgressText, setOpProgressText] = useState<string>('')

    93→    93→  useEffect(() => { currentProjectRef.current = currentProject }, [currentProject])

   111→   111→    if (photoshoot) return null // Not used in photoshoot mode

   113→   113→    return channelStudents.find((s) => s.status === targetStatus) ?? null
   114→   114→  }, [channelStudents, myChannel, photoshoot])
   115→   115→
   116→   116→  const nextPending = useMemo<Student | null>(() => {
   117→   117→    return channelStudents.find((s) => s.status === 'pending') ?? null
   118→   118→  }, [channelStudents])
   119→   119→
   120→   120→  const remainingCount = useMemo<number>(() => {
   121→   121→    return channelStudents.filter((s) => s.status === 'pending').length
   122→   122→  }, [channelStudents])
   123→   123→
   124→   124→  const isPhotographing = !photoshoot && currentlyActive !== null
   125→   125→
   126→   126→  // ── Photoshoot: students sent to operators ──────────────────────────────
   127→   127→  const sentStudents = useMemo<Student[]>(() => {
   128→   128→    if (!photoshoot) return []
   129→   129→    return channelStudents.filter((s) => s.status === 'sent')
   130→   130→  }, [photoshoot, channelStudents])
   131→   131→
   132→   132→  // ── Photoshoot: check per-channel completion from photoHistory ──────────
   133→   133→  const getStudentChannelCompletion = useCallback((studentId: string): Record<number, boolean> => {
   134→   134→    const proj = currentProjectRef.current
   135→   135→    if (!proj) return {}
   136→   136→    const result: Record<number, boolean> = {}
   137→   137→    const chCount = channelCount(proj.config.mode)
   138→   138→    for (let ch = 1; ch <= chCount; ch++) {
   139→   139→      result[ch] = proj.photoHistory.some((h) => h.student.id === studentId && h.channel === ch)
   140→   140→    }
   141→   141→    return result
   142→   142→  }, [])
   143→   143→
   144→   144→  // ── Photoshoot: filtered search results ───────────────────────────────────
   145→   145→  const searchResults = useMemo<Student[]>(() => {
   146→   146→    if (!photoshoot || !searchQuery.trim()) return []
   147→   147→    const q = searchQuery.toLowerCase().trim()
   148→   148→    return channelStudents.filter(
   149→   149→      (s) =>
   150→   150→        (s.status === 'pending' || s.status === 'sent') &&
   151→   151→        (s.nim.toLowerCase().includes(q) || s.nama.toLowerCase().includes(q))
   152→   152→    )
   153→   153→  }, [photoshoot, searchQuery, channelStudents])
   154→   154→
   155→   155→  const activeRowRef = useRef<HTMLDivElement>(null)
   156→   156→  const nextRowRef = useRef<HTMLDivElement>(null)
   157→   157→
   158→   158→  useEffect(() => {
   159→   159→    const target = activeRowRef.current ?? nextRowRef.current
   160→   160→    if (target) {
   161→   161→      target.scrollIntoView({ behavior: 'smooth', block: 'center' })

   175→   175→      setOpProgressText('')
   176→   176→      saveProjectsToStorageNow()
   177→   177→    }
   178→   178→
   179→   179→    onLocal('STUDENT_DONE', handleStudentDone)
   180→   180→    return () => { offLocal('STUDENT_DONE', handleStudentDone) }
   181→   181→  }, [updateStudentStatus, saveProjectsToStorageNow, photoshoot])
   182→   182→

   190→   190→        const mergedDb = mergeDatabases(curProj.database, proj.database)

   193→   193→          proj.photoHistory ?? [],
   194→   194→          curProj.photoHistory,
   195→   195→        )
   196→   196→        updateCurrentProject({
   197→   197→          ...curProj,
   198→   198→          database: mergedDb,
   199→   199→          photoHistory: mergedPhotoHistory,
   200→   200→          config: mergedConfig,
   201→   201→        })
   202→   202→      }
   203→   203→    }
   204→   204→
   205→   205→    onLocal('SYNC_DB', handleSyncDb)
   206→   206→    return () => { offLocal('SYNC_DB', handleSyncDb) }
   207→   207→  }, [updateCurrentProject])
   208→   208→
   209→   209→  // ── Socket: PHOTOS_SAVED
   210→   210→  // In non-photoshoot mode: STUDENT_DONE already unblocked MC, this is a no-op.
   211→   211→  // In photoshoot mode: adds photo to history and checks per-channel completion.
   212→   212→  useEffect(() => {
   213→   213→    const handlePhotosSaved = (data: PhotosSavedData) => {
   214→   214→      console.log('[SAATIRIL MC] PHOTOS_SAVED received:', data.student?.nama, 'channel:', data.channel)
   215→   215→
   216→   216→      // Non-photoshoot: already handled by STUDENT_DONE event — skip here
   217→   217→      if (!photoshoot) return

   223→   223→      const historyItem: PhotoHistoryItem = {
   224→   224→        student: data.student,
   225→   225→        photos: data.photos,
   226→   226→        channel: data.channel,
   227→   227→      }
   228→   228→
   229→   229→      const existing = curProj.photoHistory.findIndex(
   230→   230→        (h) => h.student.id === data.student.id && h.channel === data.channel,

   232→   232→      let newHistory: PhotoHistoryItem[]
   233→   233→      if (existing !== -1) {
   234→   234→        newHistory = [...curProj.photoHistory]
   235→   235→        newHistory[existing] = historyItem
   236→   236→      } else {
   237→   237→        newHistory = [...curProj.photoHistory, historyItem]
   238→   238→      }
   239→   239→
   240→   240→      // Check if all required channels have completed
   241→   241→      const chCount = channelCount(curProj.config.mode)
   242→   242→      let allChannelsDone = true
   243→   243→      for (let ch = 1; ch <= chCount; ch++) {
   244→   244→        const chDone = newHistory.some((h) => h.student.id === data.student.id && h.channel === ch)
   245→   245→        if (!chDone) {
   246→   246→          allChannelsDone = false
   247→   247→          break
   248→   248→        }
   249→   249→      }
   250→   250→
   251→   251→      const updatedProject = {
   252→   252→        ...curProj,
   253→   253→        database: curProj.database.map((s) =>
   254→   254→          s.id === data.student.id && allChannelsDone ? { ...s, status: 'done' as StudentStatus } : s
   255→   255→        ),
   256→   256→        photoHistory: newHistory,
   257→   257→      }
   258→   258→      updateCurrentProject(updatedProject)
   259→   259→      saveProjectsToStorageNow()
   260→   260→
   261→   261→      console.log('[SAATIRIL MC] PHOTOS_SAVED: allChannelsDone =', allChannelsDone, 'for', data.student.nama)
   262→   262→    }
   263→   263→
   264→   264→    onLocal('PHOTOS_SAVED', handlePhotosSaved)
   265→   265→    return () => { offLocal('PHOTOS_SAVED', handlePhotosSaved) }
   266→   266→  }, [updateStudentStatus, updateCurrentProject, saveProjectsToStorageNow, photoshoot])
   267→   267→
   268→   268→  // ── Socket: OP_PROGRESS
   269→   269→  useEffect(() => {
   270→   270→    const handleOpProgress = (data: OpProgressData) => {
   271→   271→      if (!photoshoot && data.channel !== myChannelRef.current) return
   272→   272→      console.log('[SAATIRIL MC] OP_PROGRESS:', data.status, 'channel:', data.channel)
   273→   273→      setOpProgressText(data.status)
   274→   274→      setOpProgressChannel(data.channel)
   275→   275→    }
   276→   276→
   277→   277→    onLocal('OP_PROGRESS', handleOpProgress)
   278→   278→    return () => { offLocal('OP_PROGRESS', handleOpProgress) }
   279→   279→  }, [photoshoot])
   280→   280→
   281→   281→  // ── Socket: MC_CALL
   282→   282→  useEffect(() => {
   283→   283→    const handleMcCall = (data: { student: Student; channel: number }) => {
   284→   284→      if (!photoshoot && data.channel !== myChannelRef.current) return
   285→   285→      updateStudentStatus(data.student.id, data.student.status)
   286→   286→    }
   287→   287→
   288→   288→    onLocal('MC_CALL', handleMcCall)
   289→   289→    return () => { offLocal('MC_CALL', handleMcCall) }
   290→   290→  }, [updateStudentStatus, photoshoot])
   291→   291→
   292→   292→  // ── Call action (non-photoshoot: sequential call)
   293→   293→  const handleCallNow = useCallback(() => {
   294→   294→    if (!nextPending || !currentProject) return
   295→   295→
   296→   296→    const newStatus: StudentStatus = `active_${myChannel}`
   297→   297→    updateStudentStatus(nextPending.id, newStatus)
   298→   298→    saveProjectsToStorageNow()
   299→   299→
   300→   300→    const latestProject = useSaatirilStore.getState().currentProject
   301→   301→    if (!latestProject) return
   302→   302→
   303→   303→    const updatedProject = {
   304→   304→      ...latestProject,
   305→   305→      database: latestProject.database.map((s) =>
   306→   306→        s.id === nextPending.id ? { ...s, status: newStatus } : s
   307→   307→      ),
   308→   308→    }
   309→   309→    updateCurrentProject(updatedProject)
   310→   310→    setOpProgressText('')
   311→   311→
   312→   312→    // PRIORITY: MC_CALL first (lightweight, operator gets student immediately)

   314→   314→      student: { ...nextPending, status: newStatus },
   315→   315→      channel: myChannel,
   316→   316→    })
   317→   317→    // Then SYNC_DB for consistency (now lightweight — photos stripped)
   318→   318→    emitLocal('SYNC_DB', { project: stripFrameForSync(updatedProject) })
   319→   319→  }, [
   320→   320→    nextPending,
   321→   321→    currentProject,
   322→   322→    myChannel,
   323→   323→    updateStudentStatus,
   324→   324→    updateCurrentProject,
   325→   325→    saveProjectsToStorageNow,
   326→   326→  ])
   327→   327→
   328→   328→  // ── Photoshoot: send selected student to operator(s)

   331→   331→

   333→   333→      // Send to BOTH channels
   334→   334→      const newStatus: StudentStatus = 'sent'
   335→   335→
   336→   336→      const latestProject = useSaatirilStore.getState().currentProject
   337→   337→      if (!latestProject) return
   338→   338→
   339→   339→      // Update student status to 'sent' in database
   340→   340→      const updatedProject = {
   341→   341→        ...latestProject,
   342→   342→        database: latestProject.database.map((s) =>
   343→   343→          s.id === selectedStudent.id ? { ...s, status: newStatus } : s
   344→   344→        ),
   345→   345→      }
   346→   346→
   347→   347→      updateStudentStatus(selectedStudent.id, newStatus)
   348→   348→      updateCurrentProject(updatedProject)
   349→   349→      saveProjectsToStorageNow()

   351→   351→      // Send MC_CALL to both channels FIRST (operators get student immediately)

   370→   370→        ...latestProject,
   371→   371→        database: latestProject.database.map((s) =>
   372→   372→          s.id === selectedStudent.id ? { ...s, status: newStatus } : s
   373→   373→        ),
   374→   374→      }
   375→   375→
   376→   376→      updateStudentStatus(selectedStudent.id, newStatus)
   377→   377→      updateCurrentProject(updatedProject)
   378→   378→      saveProjectsToStorageNow()
   379→   379→
   380→   380→      emitLocal('MC_CALL', {
   381→   381→        student: { ...selectedStudent, status: newStatus },
   382→   382→        channel: 1,
   383→   383→      })
   384→   384→      emitLocal('SYNC_DB', { project: stripFrameForSync(updatedProject) })
   385→   385→    }
   386→   386→
   387→   387→    setSearchQuery('')
   388→   388→    setSelectedStudent(null)
   389→   389→  }, [selectedStudent, currentProject, dualPhotoshoot, updateStudentStatus, updateCurrentProject, saveProjectsToStorageNow])
   390→   390→
   391→   391→  // ── Photoshoot: reset (for retake)
   392→   392→  const handleResetForRetake = useCallback((student: Student) => {
   393→   393→    if (!currentProject) return
   394→   394→
   395→   395→    const latestProject = useSaatirilStore.getState().currentProject
   396→   396→    if (!latestProject) return
   397→   397→
   398→   398→    const updatedProject = {
   399→   399→      ...latestProject,
   400→   400→      database: latestProject.database.map((s) =>
   401→   401→        s.id === student.id ? { ...s, status: 'pending' as StudentStatus } : s
   402→   402→      ),
   403→   403→    }
   404→   404→
   405→   405→    updateStudentStatus(student.id, 'pending')
   406→   406→    updateCurrentProject(updatedProject)
   407→   407→    saveProjectsToStorageNow()
   408→   408→
   409→   409→    emitLocal('SYNC_DB', { project: stripFrameForSync(updatedProject) })
   410→   410→
   411→   411→    // Pre-select the student for easy re-send
   412→   412→    setSelectedStudent({ ...student, status: 'pending' })
   413→   413→    setSearchQuery(student.nama)
   414→   414→  }, [currentProject, updateStudentStatus, updateCurrentProject, saveProjectsToStorageNow])
   415→   415→
   416→   416→  // ── Render helpers
   417→   417→  const renderCallButton = () => {
   418→   418→    if (readOnly) {
   419→   419→      return (
   420→   420→        <Button
   421→   421→          disabled
   422→   422→          className={`w-full font-bold cursor-not-allowed ${isMobile ? 'h-12 text-sm' : 'h-14 text-lg'}`}
   423→   423→          style={{
   424→   424→            backgroundColor: THEME.panel,
   425→   425→            color: THEME.muted,
   426→   426→            border: `2px solid ${THEME.border}`,
   427→   427→            opacity: 0.6,
   428→   428→          }}
   429→   429→        >
   430→   430→          <Monitor className="size-5" />
   431→   431→          MONITOR — HANYA LIHAT
   432→   432→        </Button>
   433→   433→      )
   434→   434→    }
   435→   435→
   436→   436→    if (photoshoot) {
   437→   437→      // Photoshoot mode: always allow sending — NO BLOCKING
   438→   438→      return (
   439→   439→        <Button
   440→   440→          disabled={!selectedStudent || selectedStudent.status === 'done'}
   441→   441→          onClick={handleSendToOperator}
   442→   442→          className={`w-full font-bold cursor-pointer transition-all duration-200 active:scale-[0.98] ${isMobile ? 'h-14 text-base' : 'h-14 text-lg hover:scale-[1.02]'}`}
   443→   443→          style={{
   444→   444→            backgroundColor: selectedStudent && selectedStudent.status !== 'done' ? THEME.emerald : THEME.panel,
   445→   445→            color: selectedStudent && selectedStudent.status !== 'done' ? THEME.bg : THEME.muted,
   446→   446→            border: `2px solid ${selectedStudent && selectedStudent.status !== 'done' ? THEME.emerald : THEME.border}`,
   447→   447→            boxShadow: selectedStudent && selectedStudent.status !== 'done' ? `0 0 20px ${THEME.emerald}44` : 'none',
