    67→export default function ProjectSetup() {
    68→  const { toast } = useToast()
    69→  const {
    70→    addProject,
    71→    setCurrentProject,
    72→    setCurrentScreen,
    73→    saveProjectsToStorageNow,
    74→  } = useSaatirilStore()
    75→
    76→  // ── Form state ─────────────────────────────────────────────────────────────
    77→  const [projectName, setProjectName] = useState('')
    78→  const [cameraMode, setCameraMode] = useState<CameraMode>('single')
    79→  const [ratio, setRatio] = useState('4:3')
    80→  const [preset, setPreset] = useState('original')
    81→  const [targetFolder, setTargetFolder] = useState('C:\\SAATIRIL_System_Out')
    82→  const [frameData, setFrameData] = useState<string | null>(null)
    83→  const [frameFileName, setFrameFileName] = useState<string>('')
    84→  const [sessionPassword, setSessionPassword] = useState('')
    85→
    86→  // Excel data: single mode uses index 0; dual uses 0 (kiri) & 1 (kanan)
    87→  const [excelData, setExcelData] = useState<[ParsedExcelData | null, ParsedExcelData | null]>([null, null])
    88→
    89→  // Loading states for file parsing
    90→  const [parsingExcel, setParsingExcel] = useState<[boolean, boolean]>([false, false])
    91→  const [parsingFrame, setParsingFrame] = useState(false)
    92→
    93→  // Refs for hidden file inputs
    94→  const excelInputRef0 = useRef<HTMLInputElement>(null)
    95→  const excelInputRef1 = useRef<HTMLInputElement>(null)
    96→  const frameInputRef = useRef<HTMLInputElement>(null)
    97→
    98→  // ── Excel parsing ──────────────────────────────────────────────────────────
    99→  const parseExcelFile = useCallback(
   100→    (file: File, channelIndex: number) => {
   101→      setParsingExcel((prev) => {
   102→        const next = [...prev] as [boolean, boolean]
   103→        next[channelIndex] = true
   104→        return next
   105→      })
   106→
   107→      const reader = new FileReader()
   108→      reader.onload = (e) => {
   109→        try {
   110→          const data = new Uint8Array(e.target?.result as ArrayBuffer)
   111→          const workbook = XLSX.read(data, { type: 'array' })
   112→          const firstSheet = workbook.Sheets[workbook.SheetNames[0]]
   113→          const jsonData = XLSX.utils.sheet_to_json<Record<string, unknown>>(firstSheet, {
   114→            defval: '',
   115→          })
   116→
   117→          if (jsonData.length === 0) {
   118→            toast({
   119→              title: 'File Kosong',
   120→              description: 'File Excel tidak memiliki data.',
   121→              variant: 'destructive',
   122→            })
   123→            setParsingExcel((prev) => {
   124→              const next = [...prev] as [boolean, boolean]
   125→              next[channelIndex] = false
   126→              return next
   127→            })
   128→            return
   129→          }
   130→
   131→          // Identify NIM and Nama columns
   132→          const headers = Object.keys(jsonData[0])
   133→          const nimCol = headers.find((h) => {
   134→            const lower = h.toLowerCase().trim()
   135→            return lower.includes('nim') || lower.includes('nis') || lower.includes('id')
   136→          })
   137→          const namaCol = headers.find((h) => {
   138→            const lower = h.toLowerCase().trim()
   139→            return lower.includes('nama') || lower.includes('name')
   140→          })
   141→
   142→          if (!nimCol || !namaCol) {
   143→            toast({
   144→              title: 'Kolom Tidak Ditemukan',
   145→              description: `Tidak dapat menemukan kolom NIM/ID dan Nama. Kolom yang ditemukan: ${headers.join(', ')}`,
   146→              variant: 'destructive',
   147→            })
   148→            setParsingExcel((prev) => {
   149→              const next = [...prev] as [boolean, boolean]
   150→              next[channelIndex] = false
   151→              return next
   152→            })
   153→            return
   154→          }
   155→
   156→          const channelNumber = channelIndex + 1
   157→          const students: Student[] = jsonData
   158→            .filter((row) => {
   159→              const nim = String(row[nimCol] ?? '').trim()
   160→              const nama = String(row[namaCol] ?? '').trim()
   161→              return nim !== '' && nama !== ''
   162→            })
   163→            .map((row) => ({
   164→              id: `ID_${Date.now()}_${Math.random().toString(36).substring(2, 9)}`,
   165→              nim: String(row[nimCol] ?? '').trim(),
   166→              nama: String(row[namaCol] ?? '').trim(),
   167→              status: 'pending' as const,
   168→              assignedChannel: channelNumber,
   169→            }))
   170→
   171→          setExcelData((prev) => {
   172→            const next = [...prev] as [ParsedExcelData | null, ParsedExcelData | null]
   173→            next[channelIndex] = { fileName: file.name, students }
   174→            return next
   175→          })
   176→
   177→          toast({
   178→            title: 'Data Berhasil Dimuat',
   179→            description: `${students.length} peserta terbaca dari "${file.name}"`,
   180→          })
   181→        } catch (err) {
   182→          console.error('[SAATIRIL] Excel parse error:', err)
   183→          toast({
   184→            title: 'Gagal Membaca File',
   185→            description: 'Pastikan file berformat .xlsx atau .xls yang valid.',
   186→            variant: 'destructive',
   187→          })
   188→        } finally {
   189→          setParsingExcel((prev) => {
   190→            const next = [...prev] as [boolean, boolean]
   191→            next[channelIndex] = false
   192→            return next
   193→          })
   194→        }
   195→      }
   196→      reader.onerror = () => {
   197→        toast({
   198→          title: 'Gagal Membaca File',
   199→          description: 'Terjadi kesalahan saat membaca file.',
   200→          variant: 'destructive',
   201→        })
   202→        setParsingExcel((prev) => {
   203→          const next = [...prev] as [boolean, boolean]
   204→          next[channelIndex] = false
   205→          return next
   206→        })
   207→      }
   208→      reader.readAsArrayBuffer(file)
   209→    },
   210→    [toast],
   211→  )
   212→
   213→  // ── Frame overlay parsing ──────────────────────────────────────────────────
   214→  const parseFrameFile = useCallback(
   215→    (file: File) => {
   216→      if (!file.name.toLowerCase().endsWith('.png')) {
   217→        toast({
   218→          title: 'Format Salah',
   219→          description: 'Frame overlay harus berformat .PNG (transparan).',
   220→          variant: 'destructive',
   221→        })
   222→        return
   223→      }
   224→
   225→      setParsingFrame(true)
   226→      const reader = new FileReader()
   227→      reader.onload = (e) => {
   228→        const dataUrl = e.target?.result as string
   229→        setFrameData(dataUrl)
   230→        setFrameFileName(file.name)
   231→        setParsingFrame(false)
   232→        toast({
   233→          title: 'Frame Dimuat',
   234→          description: `"${file.name}" berhasil dimuat sebagai overlay.`,
   235→        })
   236→      }
   237→      reader.onerror = () => {
   238→        toast({
   239→          title: 'Gagal Membaca Frame',
   240→          description: 'Terjadi kesalahan saat membaca file frame.',
   241→          variant: 'destructive',
   242→        })
   243→        setParsingFrame(false)
   244→      }
   245→      reader.readAsDataURL(file)
   246→    },
   247→    [toast],
   248→  )
   249→
   250→  // ── Drop zone handlers ─────────────────────────────────────────────────────
   251→  const handleExcelDrop = useCallback(
   252→    (e: React.DragEvent, channelIndex: number) => {
   253→      e.preventDefault()
   254→      e.stopPropagation()
   255→      const file = e.dataTransfer.files[0]
   256→      if (file) parseExcelFile(file, channelIndex)
   257→    },
   258→    [parseExcelFile],
   259→  )
   260→
   261→  const handleFrameDrop = useCallback(
   262→    (e: React.DragEvent) => {
   263→      e.preventDefault()
   264→      e.stopPropagation()
   265→      const file = e.dataTransfer.files[0]
   266→      if (file) parseFrameFile(file)
   267→    },
   268→    [parseFrameFile],
   269→  )
   270→
   271→  const handleDragOver = useCallback((e: React.DragEvent) => {
   272→    e.preventDefault()
   273→    e.stopPropagation()
   274→  }, [])
   275→
   276→  // ── Form validation ────────────────────────────────────────────────────────
   277→  const isNameValid = projectName.trim().length > 0
   278→  const isSingleDataReady = (cameraMode === 'single' || cameraMode === 'single-photoshoot' || cameraMode === 'dual-photoshoot') && excelData[0] !== null && excelData[0]!.students.length > 0
   279→  const isDualDataReady =
   280→    cameraMode === 'dual' &&
   281→    excelData[0] !== null &&
   282→    excelData[0]!.students.length > 0 &&
   283→    excelData[1] !== null &&
   284→    excelData[1]!.students.length > 0
   285→  const isDataReady = isSingleDataReady || isDualDataReady
   286→  const canStart = isNameValid && isDataReady
   287→
   288→  // ── Submit / create project ────────────────────────────────────────────────
   289→  const handleCreateProject = useCallback(() => {
   290→    if (!canStart) return
   291→
   292→    const allStudents: Student[] = []
   293→
   294→    if (cameraMode === 'dual') {
   295→      allStudents.push(...(excelData[0]?.students ?? []))
   296→      allStudents.push(...(excelData[1]?.students ?? []))
   297→    } else {
   298→      // single, single-photoshoot, dual-photoshoot: only 1 data upload
   299→      allStudents.push(...(excelData[0]?.students ?? []))
   300→    }
   301→
   302→    // Re-generate unique IDs to avoid collision from parallel parsing
   303→    const finalStudents: Student[] = allStudents.map((s, i) => ({
   304→      ...s,
   305→      id: `ID_${Date.now()}_${i}_${Math.random().toString(36).substring(2, 9)}`,
   306→    }))
   307→
   308→    const projectId = `PRJ_${Date.now()}_${Math.random().toString(36).substring(2, 9)}`
   309→
   310→    // Create subfolder based on project name under target folder
   311→    const sanitizedName = projectName.trim().replace(/[^a-zA-Z0-9_\-\s]/g, '').replace(/\s+/g, '_')
   312→    const finalTargetFolder = targetFolder
   313→      ? `${targetFolder.replace(/[\\/]+$/, '')}\\${sanitizedName}`
   314→      : `C:\\SAATIRIL_System_Out\\${sanitizedName}`
   315→
   316→    const project = {
   317→      id: projectId,
   318→      name: projectName.trim(),
   319→      config: {
   320→        mode: cameraMode,
   321→        ratio,
   322→        preset,
   323→        targetFolder: finalTargetFolder,
   324→        frame: frameData,
   325→        sessionPassword: sessionPassword.trim() || undefined,
   326→      },
   327→      database: finalStudents,
   328→      photoHistory: [],
   329→    }
   330→
   331→    addProject(project)
   332→    setCurrentProject(project)
   333→
   334→    // IMMEDIATE save before navigation to prevent data loss on first launch
   335→    saveProjectsToStorageNow()
   336→
   337→    // ── Create project folder on disk (Electron only) ────────────────────
   338→    const api = window.saatirilAPI
   339→    if (api?.createFolder) {
   340→      api.createFolder(finalTargetFolder).then((result: { success: boolean; path?: string; error?: string }) => {
   341→        if (result.success) {
   342→          console.log(`[SAATIRIL] ✅ Project folder created: ${result.path}`)
   343→          toast({
   344→            title: 'Proyek Dibuat!',
   345→            description: `"${projectName.trim()}" — ${finalStudents.length} peserta dimuat. Folder: ${finalTargetFolder}`,
   346→          })
   347→        } else {
   348→          console.error('[SAATIRIL] Failed to create project folder:', result.error)
   349→          toast({
   350→            title: 'Proyek Dibuat!',
   351→            description: `"${projectName.trim()}" — ${finalStudents.length} peserta. ⚠️ Folder gagal: ${result.error}`,
   352→            variant: 'destructive',
   353→          })
   354→        }
   355→      }).catch((err: Error) => {
   356→        console.error('[SAATIRIL] createFolder IPC error:', err)
   357→        toast({
   358→          title: 'Proyek Dibuat!',
   359→          description: `"${projectName.trim()}" — ${finalStudents.length} peserta dimuat.`,
   360→        })
   361→      })
   362→    } else {
   363→      toast({
   364→        title: 'Proyek Dibuat!',
   365→        description: `"${projectName.trim()}" — ${finalStudents.length} peserta dimuat.`,
   366→      })
   367→    }
   368→
   369→    // Sync database over LAN — strip frame data to save bandwidth
   370→    emitLocal('SYNC_DB', { project: stripFrameForSync(project) })
   371→
   372→    // Small delay to ensure state is persisted before navigation
   373→    setTimeout(() => {
   374→      setCurrentScreen('app')
   375→    }, 50)
   376→  }, [
   377→    canStart,
   378→    cameraMode,
   379→    excelData,
   380→    projectName,
   381→    ratio,
   382→    preset,
   383→    targetFolder,
   384→    frameData,
   385→    sessionPassword,
   386→    addProject,
   387→    setCurrentProject,
   388→    saveProjectsToStorageNow,
   389→    setCurrentScreen,
   390→    toast,
   391→  ])
   392→
   393→  // ── Upload zone component ──────────────────────────────────────────────────
   394→  const renderUploadZone = (
   395→    channelIndex: number,
   396→    label: string,
   397→    accentColor: string,
   398→    accentBorder: string,
   399→  ) => {
   400→    const data = excelData[channelIndex]
   401→    const isLoading = parsingExcel[channelIndex]
   402→    const inputRef = channelIndex === 0 ? excelInputRef0 : excelInputRef1
   403→
   404→    return (
   405→      <div className="space-y-2">
   406→        <Label className="text-sm font-medium" style={{ color: accentColor }}>
   407→          {label}
   408→        </Label>
   409→        <div
   410→          onDrop={(e) => handleExcelDrop(e, channelIndex)}
   411→          onDragOver={handleDragOver}
   412→          onClick={() => inputRef.current?.click()}
   413→          className={`
   414→            relative cursor-pointer rounded-lg border-2 border-dashed p-6
   415→            transition-all duration-200
   416→            hover:border-opacity-100 hover:bg-white/5
   417→            ${data ? accentBorder : 'border-[#533485]'}
   418→          `}
   419→        >
   420→          <input
   421→            ref={inputRef}
   422→            type="file"
   423→            accept=".xlsx,.xls,.csv"
   424→            className="hidden"
   425→            onChange={(e) => {
   426→              const file = e.target.files?.[0]
   427→              if (file) parseExcelFile(file, channelIndex)
   428→              // Reset so same file can be re-selected
   429→              e.target.value = ''
   430→            }}
   431→          />
   432→
   433→          <div className="flex flex-col items-center gap-2 text-center">
   434→            {isLoading ? (
   435→              <>
   436→                <div className="h-8 w-8 animate-spin rounded-full border-2 border-[#533485] border-t-[#d4af37]" />
   437→                <p className="text-xs text-[#c4b5fd]">Membaca file...</p>
   438→              </>
   439→            ) : data ? (
   440→              <>
   441→                <FileSpreadsheet className="h-8 w-8" style={{ color: accentColor }} />
   442→                <p className="text-sm font-medium" style={{ color: accentColor }}>
   443→                  Data siap: {data.students.length} Peserta Terbaca
   444→                </p>
   445→                <p className="text-xs text-[#c4b5fd]">{data.fileName}</p>
   446→                <button
   447→                  onClick={(e) => {
   448→                    e.stopPropagation()
   449→                    setExcelData((prev) => {
   450→                      const next = [...prev] as [ParsedExcelData | null, ParsedExcelData | null]
   451→                      next[channelIndex] = null
   452→                      return next
   453→                    })
   454→                  }}
   455→                  className="mt-1 rounded-full p-1 text-[#c4b5fd] transition-colors hover:bg-white/10 hover:text-red-400"
   456→                >
   457→                  <X className="h-4 w-4" />
   458→                </button>
   459→              </>
   460→            ) : (
   461→              <>
   462→                <Upload className="h-8 w-8 text-[#533485]" />
   463→                <p className="text-sm text-[#c4b5fd]">
   464→                  Drag & drop file Excel di sini
   465→                </p>
   466→                <p className="text-xs text-[#533485]">.xlsx / .xls / .csv</p>
   467→              </>
   468→            )}
   469→          </div>
   470→        </div>
   471→      </div>
   472→    )
   473→  }
   474→
   475→  // ── Render ─────────────────────────────────────────────────────────────────
   476→  return (
   477→    <div className="relative h-screen bg-[#1a0b2e] backdrop-blur-sm overflow-hidden">
   478→      {/* Subtle gradient overlay */}
   479→      <div className="pointer-events-none absolute inset-0 bg-gradient-to-br from-[#1a0b2e] via-[#2a164a]/30 to-[#1a0b2e]" />
   480→
   481→      <div className="relative z-10 flex h-full flex-col">
   482→        {/* ── Header ─────────────────────────────────────────────────────────── */}
   483→        <header className="flex items-center gap-3 border-b border-[#533485]/50 px-5 py-2.5">
   484→          <Button
   485→            variant="ghost"
   486→            size="icon"
   487→            onClick={() => setCurrentScreen('hub')}
   488→            className="text-[#c4b5fd] hover:bg-white/10 hover:text-[#d4af37]"
   489→          >
   490→            <ArrowLeft className="h-4 w-4" />
   491→          </Button>
   492→          <div>
   493→            <h1 className="text-base font-bold text-white">
   494→              <span className="text-[#d4af37]">SAATIRIL</span> — Buat Proyek Baru
   495→            </h1>
   496→          </div>
   497→        </header>
   498→
   499→        {/* ── Main Content: 2-Column Layout ──────────────────────────────────── */}
   500→        <main className="flex-1 overflow-y-auto p-4">
   501→          <div className="mx-auto grid max-w-6xl gap-4 lg:grid-cols-2">
   502→            {/* ── LEFT COLUMN ──────────────────────────────────────────────── */}
   503→            <div className="flex flex-col gap-3">
   504→              {/* Project Name */}
   505→              <Card className="border-[#533485] bg-[#2a164a] shadow-lg">
   506→                <CardHeader className="pb-2">
   507→                  <CardTitle className="flex items-center gap-2 text-base text-white">
   508→                    <Monitor className="h-5 w-5 text-[#d4af37]" />
   509→                    Nama Proyek
   510→                  </CardTitle>
   511→                </CardHeader>
   512→                <CardContent>
   513→                  <Input
   514→                    placeholder="Masukkan nama event / proyek..."
   515→                    value={projectName}
   516→                    onChange={(e) => setProjectName(e.target.value)}
   517→                    className="border-[#533485] bg-[#3b2263] text-white placeholder:text-[#533485] focus-visible:border-[#d4af37] focus-visible:ring-[#d4af37]/30"
   518→                  />
   519→                </CardContent>
   520→              </Card>
   521→
   522→              {/* Session Password (Keamanan LAN) */}
   523→              <Card className="border-[#533485] bg-[#2a164a] shadow-lg">
   524→                <CardHeader className="pb-2">
   525→                  <CardTitle className="flex items-center gap-2 text-base text-white">
   526→                    <svg className="h-5 w-5 text-[#d4af37]" xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><rect width="18" height="11" x="3" y="11" rx="2" ry="2"/><path d="M7 11V7a5 5 0 0 1 10 0v4"/></svg>
   527→                    Password Sesi LAN
