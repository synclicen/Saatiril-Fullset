    27→// ─── Helpers ────────────────────────────────────────────────────────────────
    28→function isActiveStatus(status: StudentStatus): boolean {
    29→  return status.startsWith('active')
    30→}
    31→
    32→function getActiveChannel(status: StudentStatus): number | null {
    33→  if (!isActiveStatus(status)) return null
    34→  const ch = status.split('_')[1]
    35→  return ch ? parseInt(ch, 10) : null
    36→}
    37→
    38→function statusLabel(status: StudentStatus): string {
    39→  if (status === 'pending') return 'Menunggu'
    40→  if (status === 'sent') return 'Dikirim'
    41→  if (status === 'done') return 'Selesai'
    42→  const ch = getActiveChannel(status)
    43→  return ch != null ? `Foto Ch.${ch}` : 'Aktif'
    44→}

   338→

   341→        ...latestProject,
   342→        database: latestProject.database.map((s) =>
   343→          s.id === selectedStudent.id ? { ...s, status: newStatus } : s
   344→        ),
   345→      }

   348→      updateCurrentProject(updatedProject)
   349→      saveProjectsToStorageNow()
   350→
   351→      // Send MC_CALL to both channels FIRST (operators get student immediately)

   353→        student: { ...selectedStudent, status: newStatus, assignedChannel: 1 },
   354→        channel: 1,

   356→      emitLocal('MC_CALL', {
   357→        student: { ...selectedStudent, status: newStatus, assignedChannel: 2 },

   366→      const latestProject = useSaatirilStore.getState().currentProject

   368→

   372→          s.id === selectedStudent.id ? { ...s, status: newStatus } : s

   377→      updateCurrentProject(updatedProject)
   378→      saveProjectsToStorageNow()

   381→        student: { ...selectedStudent, status: newStatus },

   511→  const getRowStyle = (student: Student): React.CSSProperties => {

   514→    const isDone = student.status === 'done'

   520→        backgroundColor: `${THEME.gold}22`,
