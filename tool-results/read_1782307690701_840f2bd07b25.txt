   113→   113→// ─── Ratio parser ───────────────────────────────────────────────────────────
   114→   114→function parseRatio(ratioStr: string): number {
   115→   115→  const parts = ratioStr.split(':')
   116→   116→  if (parts.length === 2) {
   117→   117→    const w = parseFloat(parts[0])
   118→   118→    const h = parseFloat(parts[1])
   119→   119→    if (w > 0 && h > 0) return w / h
   120→   120→  }
   121→   121→  return 4 / 3
   122→   122→}

   157→   157→function fitAspectRatio(
   158→   158→  availW: number,
   159→   159→  availH: number,
   160→   160→  ratio: number,
   161→   161→): { width: number; height: number } {
   162→   162→  const hFromW = availW / ratio
   163→   163→  if (hFromW <= availH) {
   164→   164→    return { width: availW, height: hFromW }
   165→   165→  }
   166→   166→  const wFromH = availH * ratio
   167→   167→  return { width: wFromH, height: availH }
   168→   168→}
